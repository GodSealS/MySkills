### Rigid body and collision
```typescript
import { RigidBody, BoxCollider, PhysicsSystem, Vec3, geometry } from 'cc';

// Add rigid body
const rigidBody = node.addComponent(RigidBody);
rigidBody.type = RigidBody.Type.DYNAMIC;
rigidBody.mass = 1.0;

// Add collider
const collider = node.addComponent(BoxCollider);
collider.size = new Vec3(1, 1, 1);
collider.on('onCollisionEnter', (event) => {
    console.log('Collision detected:', event.otherCollider.node.name);
});

// Raycast
const ray = new geometry.Ray(origin.x, origin.y, origin.z, direction.x, direction.y, direction.z);
const physics = PhysicsSystem.instance;
if (physics.raycastClosest(ray)) {
    const hit = physics.raycastClosestResult;
    console.log('Hit:', hit.collider.node.name, hit.distance);
}
```
